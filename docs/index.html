from wpipe import Pipeline, Condition, Parallel
from dto.context import PipelineContextDTO
from wpipe_steps.connectivity import HttpRequest
from states.ai_sentiment_step import AISentimentStep


def build_pipeline() -> Pipeline:
    """Generated by WPipe Studio from order_risk_pipeline.wpipe"""
    pipe = Pipeline(verbose=True)
    pipe.set_steps([
        HttpRequest(),
        AISentimentStep(threshold=0.8),
        Condition(
            expression="risk > 0.8",
            branch_true=[
                Parallel(
                    steps=[MySQLQuery(), SlackNotify()],
                    merge_policy="accumulate",
                ),
            ],
        ),
    ])
    return pipe
